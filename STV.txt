    <script>
        // Variables globales
        let transports = JSON.parse(localStorage.getItem('transports')) || [];
        let deferredPrompt;

        // Elementos DOM
        const transportList = document.getElementById('transportList');
        const transportModal = document.getElementById('transportModal');
        const transportForm = document.getElementById('transportForm');
        const modalTitle = document.getElementById('modalTitle');
        const cancelModalBtn = document.getElementById('cancelModalBtn');
        const newTransportBtn = document.getElementById('newTransportBtn');
        const exportBtn = document.getElementById('exportBtn');
        const importBtn = document.getElementById('importBtn');
        const importFile = document.getElementById('importFile');
        const backupBtn = document.getElementById('backupBtn');
        const restoreBtn = document.getElementById('restoreBtn');
        const notification = document.getElementById('notification');
        const notificationText = document.getElementById('notificationText');
        const installPrompt = document.getElementById('installPrompt');
        const installBtn = document.getElementById('installBtn');
        const tabs = document.querySelectorAll('.tab');
        const tabContents = document.querySelectorAll('.tab-content');

        // Inicialización
        document.addEventListener('DOMContentLoaded', () => {
            renderTransports();
            setupEventListeners();
            checkPWAInstallable();
        });

        // Configurar event listeners
        function setupEventListeners() {
            // Formulario de transporte
            transportForm.addEventListener('submit', handleTransportSubmit);
            cancelModalBtn.addEventListener('click', closeModal);
            newTransportBtn.addEventListener('click', () => openModal());
            
            // Copias de seguridad
            exportBtn.addEventListener('click', exportData);
            importBtn.addEventListener('click', () => importFile.click());
            importFile.addEventListener('change', importData);
            backupBtn.addEventListener('click', exportData);
            restoreBtn.addEventListener('click', () => importFile.click());
            
            // PWA installation
            window.addEventListener('beforeinstallprompt', (e) => {
                e.preventDefault();
                deferredPrompt = e;
                installPrompt.style.display = 'flex';
            });
            
            installBtn.addEventListener('click', async () => {
                if (deferredPrompt) {
                    deferredPrompt.prompt();
                    const { outcome } = await deferredPrompt.userChoice;
                    if (outcome === 'accepted') {
                        installPrompt.style.display = 'none';
                    }
                    deferredPrompt = null;
                }
            });
            
            // Tabs
            tabs.forEach(tab => {
                tab.addEventListener('click', () => {
                    const tabId = tab.getAttribute('data-tab');
                    activateTab(tabId);
                });
            });
        }

        // Funciones de la interfaz
        function renderTransports() {
            transportList.innerHTML = '';
            
            if (transports.length === 0) {
                transportList.innerHTML = `
                    <div style="grid-column: 1 / -1; text-align: center; padding: 40px; color: #777;">
                        <i class="fas fa-ship" style="font-size: 3rem; margin-bottom: 15px;"></i>
                        <p>No hay transportes registrados. Agrega el primero haciendo clic en "Nuevo Transporte".</p>
                    </div>
                `;
                return;
            }
            
            transports.forEach(transport => {
                const transportCard = document.createElement('div');
                transportCard.className = 'transport-card';
                transportCard.innerHTML = `
                    <div class="transport-name">${transport.name}</div>
                    <div class="transport-detail">
                        <span>Tipo:</span>
                        <span>${getTransportTypeLabel(transport.type)}</span>
                    </div>
                    <div class="transport-detail">
                        <span>Capacidad:</span>
                        <span>${transport.capacity} personas</span>
                    </div>
                    <div class="transport-detail">
                        <span>Horario:</span>
                        <span>${transport.schedule || 'No especificado'}</span>
                    </div>
                    <div class="transport-detail">
                        <span>Contacto:</span>
                        <span>${transport.contact || 'No especificado'}</span>
                    </div>
                    <div class="transport-actions">
                        <button class="action-btn edit-btn" data-id="${transport.id}">
                            <i class="fas fa-edit"></i>
                        </button>
                        <button class="action-btn delete-btn" data-id="${transport.id}">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
                `;
                
                transportList.appendChild(transportCard);
            });
            
            // Agregar event listeners a los botones de editar y eliminar
            document.querySelectorAll('.edit-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    const id = e.currentTarget.getAttribute('data-id');
                    const transport = transports.find(t => t.id === id);
                    if (transport) openModal(transport);
                });
            });
            
            document.querySelectorAll('.delete-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    const id = e.currentTarget.getAttribute('data-id');
                    deleteTransport(id);
                });
            });
        }
        
        function getTransportTypeLabel(type) {
            const types = {
                'lancha': 'Lancha',
                'ferry': 'Ferry',
                'bote': 'Bote',
                'helicoptero': 'Helicóptero'
            };
            return types[type] || type;
        }
        
        function openModal(transport = null) {
            if (transport) {
                // Modo edición
                modalTitle.textContent = 'Editar Transporte';
                document.getElementById('transportId').value = transport.id;
                document.getElementById('transportName').value = transport.name;
                document.getElementById('transportType').value = transport.type;
                document.getElementById('capacity').value = transport.capacity;
                document.getElementById('schedule').value = transport.schedule || '';
                document.getElementById('contact').value = transport.contact || '';
            } else {
                // Modo nuevo
                modalTitle.textContent = 'Nuevo Transporte';
                transportForm.reset();
                document.getElementById('transportId').value = '';
            }
            
            transportModal.style.display = 'flex';
        }
        
        function closeModal() {
            transportModal.style.display = 'none';
        }
        
        function handleTransportSubmit(e) {
            e.preventDefault();
            
            const transportId = document.getElementById('transportId').value;
            const transportData = {
                name: document.getElementById('transportName').value,
                type: document.getElementById('transportType').value,
                capacity: parseInt(document.getElementById('capacity').value),
                schedule: document.getElementById('schedule').value,
                contact: document.getElementById('contact').value
            };
            
            if (transportId) {
                // Editar transporte existente
                transportData.id = transportId;
                updateTransport(transportData);
            } else {
                // Crear nuevo transporte
                transportData.id = Date.now().toString();
                addTransport(transportData);
            }
            
            closeModal();
        }
        
        function addTransport(transport) {
            transports.push(transport);
            saveTransports();
            showNotification('Transporte agregado con éxito');
            renderTransports();
        }
        
        function updateTransport(transportData) {
            const index = transports.findIndex(t => t.id === transportData.id);
            if (index !== -1) {
                transports[index] = transportData;
                saveTransports();
                showNotification('Transporte actualizado con éxito');
                renderTransports();
            }
        }
        
        function deleteTransport(id) {
            if (confirm('¿Estás seguro de que deseas eliminar este transporte?')) {
                transports = transports.filter(t => t.id !== id);
                saveTransports();
                showNotification('Transporte eliminado con éxito');
                renderTransports();
            }
        }
        
        function saveTransports() {
            localStorage.setItem('transports', JSON.stringify(transports));
        }
        
        // Funciones de copia de seguridad
        function exportData() {
            const dataStr = JSON.stringify(transports, null, 2);
            const dataUri = 'data:application/json;charset=utf-8,'+ encodeURIComponent(dataStr);
            
            const exportFileDefaultName = `transporte-isla-lobos-${new Date().toISOString().split('T')[0]}.json`;
            
            const linkElement = document.createElement('a');
            linkElement.setAttribute('href', dataUri);
            linkElement.setAttribute('download', exportFileDefaultName);
            linkElement.click();
            
            showNotification('Datos exportados con éxito');
        }
        
        function importData(e) {
            const file = e.target.files[0];
            if (!file) return;
            
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const importedData = JSON.parse(e.target.result);
                    if (Array.isArray(importedData)) {
                        if (confirm('¿Estás seguro de que deseas importar estos datos? Se sobrescribirán los datos actuales.')) {
                            transports = importedData;
                            saveTransports();
                            renderTransports();
                            showNotification('Datos importados con éxito');
                        }
                    } else {
                        throw new Error('Formato de archivo inválido');
                    }
                } catch (error) {
                    alert('Error al importar el archivo: ' + error.message);
                }
            };
            reader.readAsText(file);
            
            // Reset input file
            e.target.value = '';
        }
        
        // Funciones de UI
        function showNotification(message) {
            notificationText.textContent = message;
            notification.classList.add('show');
            
            setTimeout(() => {
                notification.classList.remove('show');
            }, 3000);
        }
        
        function activateTab(tabId) {
            // Activar tab
            tabs.forEach(tab => {
                if (tab.getAttribute('data-tab') === tabId) {
                    tab.classList.add('active');
                } else {
                    tab.classList.remove('active');
                }
            });
            
            // Mostrar contenido del tab
            tabContents.forEach(content => {
                if (content.id === `${tabId}Tab`) {
                    content.classList.add('active');
                } else {
                    content.classList.remove('active');
                }
            });
        }
        
        function checkPWAInstallable() {
            if (window.matchMedia('(display-mode: standalone)').matches) {
                installPrompt.style.display = 'none';
            }
        }

        // Service Worker Registration
        if ('serviceWorker' in navigator) {
            window.addEventListener('load', () => {
                navigator.serviceWorker.register('sw.js').then(registration => {
                    console.log('SW registered: ', registration);
                }).catch(registrationError => {
                    console.log('SW registration failed: ', registrationError);
                });
            });
        }
    </script>

    <!-- Service Worker (sw.js would be a separate file, but including inline for simplicity) -->
    <script>
        // Este código normalmente estaría en un archivo sw.js separado
        const CACHE_NAME = 'transporte-isla-lobos-v1';
        const urlsToCache = [
            '/',
            '/styles/main.css',
            '/script/main.js'
        ];

        self.addEventListener('install', event => {
            event.waitUntil(
                caches.open(CACHE_NAME)
                    .then(cache => {
                        return cache.addAll(urlsToCache);
                    })
            );
        });

        self.addEventListener('fetch', event => {
            event.respondWith(
                caches.match(event.request)
                    .then(response => {
                        if (response) {
                            return response;
                        }
                        return fetch(event.request);
                    }
                )
            );
        });
    </script>
